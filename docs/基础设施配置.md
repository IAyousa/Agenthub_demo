## 2. Spring Boot 配置

### 2.1 主配置文件
```bash
# backend-java/src/main/resources/application.yml
# 公共配置（所有环境生效）

server:
  port: 8080

spring:
  application:
    name: agenthub

  # 数据源配置
  datasource:
    url: jdbc:postgresql://localhost:5432/agenthub
    username: agenthub
    password: agenthub123
    driver-class-name: org.postgresql.Driver

  # JPA 配置
  jpa:
    hibernate:
      ddl-auto: update               # 开发环境自动建表
    show-sql: false                   # 生产环境关闭 SQL 日志
    properties:
      hibernate:
        format_sql: true
        dialect: org.hibernate.dialect.PostgreSQLDialect

  # 文件上传大小限制
  servlet:
    multipart:
      max-file-size: 10MB
      max-request-size: 10MB

# Agent 服务地址
agent:
  service:
    url: http://localhost:8000
    timeout: 120000                   # 超时时间（毫秒）

# 产物存储路径（MVP 使用本地文件系统）
artifact:
  storage:
    path: ./artifacts
    url-prefix: /artifacts
```
### 2.2 开发环境配置
```bash
# backend-java/src/main/resources/application-dev.yml
# 开发环境专属配置

spring:
```

## 3. FastAPI 配置
### 3.1 应用配置
```bash
# agent-service/config.py
# 应用配置（所有环境变量和常量集中管理）

import os

class Settings:
    # ========== 服务配置 ==========
    APP_TITLE = "AgentHub Agent Service"
    APP_VERSION = "1.0.0"
    HOST = "0.0.0.0"
    PORT = 8000

    # ========== Claude API 配置 ==========
    CLAUDE_API_KEY = os.getenv("CLAUDE_API_KEY", "your-api-key-here")
    CLAUDE_API_URL = "https://api.anthropic.com/v1/messages"
    CLAUDE_MODEL = "claude-3-5-sonnet-20241022"

    # ========== Codex API 配置 ==========
    CODEX_API_KEY = os.getenv("CODEX_API_KEY", "your-api-key-here")
    CODEX_API_URL = "https://api.openai.com/v1/chat/completions"
    CODEX_MODEL = "gpt-4o"

    # ========== Agent 调用参数 ==========
    AGENT_TIMEOUT = 120               # 超时（秒）
    AGENT_MAX_TOKENS = 4096           # 最大输出 token 数

    # ========== 安全配置 ==========
    ALLOWED_ORIGINS = [
        "http://localhost:5173",      # Vue 开发服务器
        "http://localhost:8080",      # Spring Boot
    ]

settings = Settings()
```

### 4.2 Vite 配置
```bash
// frontend/vite.config.js
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import path from 'path'

export default defineConfig({
  plugins: [vue()],

  resolve: {
    alias: {
      '@': path.resolve(__dirname, './src')
    }
  },

  server: {
    port: 5173,
    proxy: {
      // 开发环境代理 API 请求到后端
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true
      },
      '/ws': {
        target: 'http://localhost:8080',
        ws: true
      }
    }
  }
})
```
